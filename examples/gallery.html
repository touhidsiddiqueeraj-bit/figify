How to generate (agent)

Copy-paste for your agent
import figtweak
fig, ax = plt.subplots(figsize=figtweak.ieee_single())
ax.plot(x, y, label="data")
ax.legend()
# before save, enforce:
print(figtweak.lint(fig))  # [] if ok
figtweak.fix(fig, template="ieee-access-single")
figtweak.save(fig, "fig.svg", template="ieee-access-single")
# or via MCP: fix_py(code, template="ieee-access-single")
